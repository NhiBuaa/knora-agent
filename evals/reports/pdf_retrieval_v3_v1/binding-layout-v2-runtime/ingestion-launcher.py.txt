def checkpoint_json(path, value):
    temporary = path.with_suffix(".tmp")
    with temporary.open("w", encoding="utf-8") as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())
    os.replace(temporary, path)


def load_json(path):
    return json.loads(Path(path).read_text(encoding="utf-8-sig"))


class IngestionGuard(FreeOnlyGuard):
    def __init__(self, path, *, run_identity, preflight, expected):
        super().__init__(path, run_identity=run_identity, preflight=preflight)
        self.expected, self.source = expected, None
        self.state.setdefault("embedding_http_attempts", 0)
        self.state.setdefault("attempted_sources", [])
        self.save()

    def save(self):
        self.path.parent.mkdir(parents=True, exist_ok=True)
        checkpoint_json(self.path, self.state)

    def stop(self, reason):
        self.state["stop_reason"] = self.state["stop_reason"] or reason
        self.save()
        raise RuntimeError(self.state["stop_reason"])

    def before_request(self, request):
        if self.state["stop_reason"]:
            raise RuntimeError(self.state["stop_reason"])
        if datetime.now(UTC) - datetime.fromisoformat(self.preflight["checked_at"]) > timedelta(
            minutes=15
        ):
            self.stop("FREE_PREFLIGHT_EXPIRED")
        endpoint = f"https://api.cloudflare.com/client/v4/accounts/{self.account_id}/ai/run/@cf/qwen/qwen3-embedding-0.6b"
        try:
            payload = json.loads(request.content)
        except (ValueError, UnicodeError):
            self.stop("EMBEDDING_REQUEST_INVALID")
        texts = self.expected.get(self.source)
        if (
            request.method != "POST"
            or str(request.url) != endpoint
            or set(payload) != {"documents"}
            or not texts
            or payload["documents"] != [unicodedata.normalize("NFKC", text) for text in texts]
            or any(key.startswith("cf-aig") for key in request.headers)
        ):
            self.stop("INGESTION_REQUEST_NOT_APPROVED")
        if self.source in self.state["attempted_sources"]:
            self.stop("SOURCE_ATTEMPT_ALREADY_CONSUMED")
        if (
            self.state["embedding_http_attempts"] >= 14
            or self.state["embedding_inputs"] + len(texts) > 47
        ):
            self.stop("INGESTION_REQUEST_LIMIT")
        self.state["embedding_http_attempts"] += 1
        self.state["attempted_sources"].append(self.source)
        self.save()
        super().before_request(request)

    def after_response(self, response):
        super().after_response(response)
        if response.status_code >= 300:
            self.state["stop_reason"] = self.state["stop_reason"] or "EMBEDDING_HTTP_FAILED"
            self.save()


class DocumentOnlyProvider:
    def __init__(self, provider, guard):
        self.provider, self.guard = provider, guard

    def embed_documents(self, texts, configuration):
        if texts != self.guard.expected.get(self.guard.source):
            self.guard.stop("EXTRACTION_CONTENT_OR_ORDER_DRIFT")
        try:
            return self.provider.embed_documents(texts, configuration)
        except Exception:
            self.guard.state["stop_reason"] = (
                self.guard.state["stop_reason"] or "EMBEDDING_PROVIDER_FAILED"
            )
            self.guard.save()
            raise

    def embed_queries(self, *args, **kwargs):
        self.guard.stop("QUERY_EMBEDDING_FORBIDDEN")

    def embed(self, *args, **kwargs):
        self.guard.stop("UNTYPED_EMBEDDING_FORBIDDEN")

    async def generate(self, *args, **kwargs):
        self.guard.stop("GENERATION_FORBIDDEN")
